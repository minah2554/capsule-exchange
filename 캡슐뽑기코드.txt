function shufflePrizes() {
  for (let i = PRIZES.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));

  }

    if(r < p.weight) return p;
    r -= p.weight;

}

function audioCtx(){ if(!actx) actx = new (window.AudioContext||window.webkitAudioContext)(); return actx; }

function playSuspense(){
  const ctx = audioCtx();
  const dur = 1.6;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = 'sawtooth';
  osc.frequency.setValueAtTime(140, ctx.currentTime);
  osc.frequency.exponentialRampToValueAtTime(420, ctx.currentTime+dur);
  gain.gain.setValueAtTime(0.0001, ctx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.09, ctx.currentTime+0.2);
  for(let t=0.2;t<dur;t+=0.12){
    gain.gain.exponentialRampToValueAtTime(0.11, ctx.currentTime+t);
    gain.gain.exponentialRampToValueAtTime(0.05, ctx.currentTime+t+0.06);
  }
  gain.gain.exponentialRampToValueAtTime(0.14, ctx.currentTime+dur);
  osc.connect(gain); gain.connect(ctx.destination);
  osc.start(); osc.stop(ctx.currentTime+dur);
}

// Draw a beautiful star shape with glow
function drawStar(ctx, cx, cy, spikes, outerRadius, innerRadius, color, alpha) {
  let rot = Math.PI / 2 * 3;
  let x = cx;
  let y = cy;
  let step = Math.PI / spikes;

  ctx.save();
  ctx.beginPath();
  ctx.moveTo(cx, cy - outerRadius);
  for (let i = 0; i < spikes; i++) {
    x = cx + Math.cos(rot) * outerRadius;
    y = cy + Math.sin(rot) * outerRadius;
    ctx.lineTo(x, y);
    rot += step;

    x = cx + Math.cos(rot) * innerRadius;
    y = cy + Math.sin(rot) * innerRadius;
    ctx.lineTo(x, y);
    rot += step;
  }
  ctx.lineTo(cx, cy - outerRadius);
  ctx.closePath();
  ctx.fillStyle = color;
  ctx.globalAlpha = alpha;
  
  // Outer glowing aura
  ctx.shadowColor = color;
  ctx.shadowBlur = 8;
  
  ctx.fill();
  ctx.restore();
}

function startCanvasLoop() {
  if (canvasLoopActive) return;
  canvasLoopActive = true;
  
  const canvas = document.getElementById('fx');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  
  function updateFrame() {
    if (!canvasLoopActive) return;
    
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    // Spawn ambient floating sparkles around/behind the card
    if (Math.random() < 0.12) {
      const colors = ['#ffd66b', '#ffffff', '#c9b6ff', '#eaf6ff'];
      particles.push({
        type: 'sparkle',
        x: 40 + Math.random() * (canvas.width - 80),
        y: canvas.height * 0.65 + (Math.random() - 0.5) * 60,
        vx: (Math.random() - 0.5) * 1.5,
        vy: -0.8 - Math.random() * 1.5,
        size: 3 + Math.random() * 4,
        color: colors[Math.floor(Math.random() * colors.length)],
        rot: Math.random() * Math.PI,
        vr: (Math.random() - 0.5) * 0.05,
        life: 0,
        maxLife: 120 + Math.random() * 60
      });
    }
    
    // Update and draw all particles
    for (let i = particles.length - 1; i >= 0; i--) {
      const p = particles[i];
      p.life++;
      
      if (p.life >= p.maxLife) {
        particles.splice(i, 1);
        continue;
      }
      
      // Burst elements fall, ambient sparkles float up
      if (p.type === 'rect' || p.type === 'star') {
        p.vy += 0.32;
        p.vx *= 0.98;
      } else {
        p.vx += Math.sin(p.life * 0.05) * 0.02; // Sway
      }
      
      p.x += p.vx;
      p.y += p.vy;
      p.rot += p.vr;
      
      const alpha = Math.max(0, 1 - (p.life / p.maxLife));
      
      if (p.type === 'rect') {
        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(p.rot);
        ctx.globalAlpha = alpha;
        ctx.fillStyle = p.color;
        ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
        ctx.restore();
      } else if (p.type === 'star') {
        drawStar(ctx, p.x, p.y, 5, p.size, p.size * 0.4, p.color, alpha);
      } else if (p.type === 'sparkle') {
        drawStar(ctx, p.x, p.y, 4, p.size, p.size * 0.25, p.color, alpha);
      }
    }
    
    canvasAnimId = requestAnimationFrame(updateFrame);
  }
  
  canvasAnimId = requestAnimationFrame(updateFrame);
}

function stopCanvasLoop() {
  canvasLoopActive = false;
  if (canvasAnimId) {
    cancelAnimationFrame(canvasAnimId);
    canvasAnimId = null;
  }
  particles = [];
  const canvas = document.getElementById('fx');
  if (canvas) {
    const ctx = canvas.getContext('2d');
    ctx.clearRect(0, 0, canvas.width, canvas.height);
  }
}

========================================================= */
let gateTaps = 0, gateTimer = null;
document.getElementById('lockLabel').addEventListener('click', function(){
  gateTaps++;
  clearTimeout(gateTimer);
  gateTimer = setTimeout(()=>{ gateTaps = 0; }, 2200);
  if(gateTaps >= 5){
    gateTaps = 0;
    dropCoin();
  }
});

function dropCoin(){
  const plate = document.querySelector('.machineSlotPlate');
  const coin = document.createElement('div');
  coin.className = 'coin drop';
  coin.textContent = '🪙';
  plate.appendChild(coin);
  playCoin();
  vibrate(30);
  coin.addEventListener('animationend', ()=>{
    coin.remove();
    unlockApp();
  });
}

function unlockApp(){
  document.getElementById('lockScreen').style.display='none';
  document.getElementById('app').style.display='flex';
  showScreen('screenMain');
  vibrate([20,30,20]);
}
